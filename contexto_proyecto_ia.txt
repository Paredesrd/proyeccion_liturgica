import sys
from PyQt6.QtWidgets import QApplication, QMainWindow, QVBoxLayout, QWidget
from PyQt6.QtWebEngineWidgets import QWebEngineView
from PyQt6.QtWebEngineCore import QWebEngineSettings, QWebEngineProfile
from PyQt6.QtCore import QUrl, Qt

from core.config import settings
from core.logger import log


class LiturgicalWindow(QMainWindow):
    def __init__(self, url: str, title: str, is_fullscreen: bool = False, target_screen=None):
        super().__init__()
        self.setWindowTitle(title)
        self.is_fullscreen_mode = is_fullscreen

        central_widget = QWidget()
        self.setCentralWidget(central_widget)
        layout = QVBoxLayout(central_widget)
        layout.setContentsMargins(0, 0, 0, 0)

        self.web_view = QWebEngineView()
        layout.addWidget(self.web_view)

        # Optimización de caché: Forzar uso de RAM en lugar de disco
        # Evita la degradación del SSD al reproducir videos de fondo de forma continua.
        profile = self.web_view.page().profile()
        profile.setHttpCacheType(QWebEngineProfile.HttpCacheType.MemoryHttpCache)

        # Configuraciones críticas de reproducción y seguridad del contenedor
        web_settings = self.web_view.page().settings()
        web_settings.setAttribute(QWebEngineSettings.WebAttribute.PlaybackRequiresUserGesture, False)
        web_settings.setAttribute(QWebEngineSettings.WebAttribute.LocalContentCanAccessRemoteUrls, True)

        self.web_view.setUrl(QUrl(url))

        if target_screen:
            self.setScreen(target_screen)
            self.move(target_screen.geometry().topLeft())

        if self.is_fullscreen_mode:
            self.setWindowFlags(Qt.WindowType.FramelessWindowHint)
            self.showFullScreen()
        else:
            self.resize(1280, 720)
            self.show()

    def keyPressEvent(self, event):
        # Mitigación de riesgo: ESC desactiva el fullscreen para recuperar control,
        # pero NUNCA apaga el proceso principal. El cierre se hace desde la [X] nativa.
        if event.key() == Qt.Key.Key_Escape and self.is_fullscreen_mode:
            self.showNormal()
            self.is_fullscreen_mode = False
            log.info(f"Modo ventana activado manualmente mediante ESC para la ventana: {self.windowTitle()}")


def run_desktop_app():
    # Inyección de banderas Chromium de alto rendimiento ANTES de instanciar la interfaz
    sys.argv.extend([
        "--enable-gpu-rasterization",
        "--ignore-gpu-blocklist",
        "--enable-zero-copy",
        "--disable-software-rasterizer"
    ])

    app = QApplication.instance()
    if app is None:
        app = QApplication(sys.argv)

    app.setQuitOnLastWindowClosed(True)

    screens = app.screens()
    num_screens = len(screens)

    if num_screens > 1:
        log.info(f"Topología detectada: {num_screens} monitores. Desplegando en Producción.")
        control_screen = screens[0]
        projector_screen = screens[1]
        projector_fullscreen = True
    else:
        log.warning("Topología detectada: 1 monitor. Desplegando en Desarrollo (Modo Ventana).")
        control_screen = screens[0]
        projector_screen = screens[0]
        projector_fullscreen = False

    control_win = LiturgicalWindow(
        url=f"http://127.0.0.1:{settings.PORT}/",
        title="Panel de Control",
        is_fullscreen=False,
        target_screen=control_screen
    )

    projector_win = LiturgicalWindow(
        url=f"http://127.0.0.1:{settings.PORT}/projector",
        title="Proyector",
        is_fullscreen=projector_fullscreen,
        target_screen=projector_screen
    )

    # Mantener referencias para evitar la recolección de basura (Garbage Collector)
    app.control_win = control_win
    app.projector_win = projector_win

    return app.exec()

def get_local_ip():
    """Obtiene la IP local de forma pasiva y sin requerir Internet."""
    try:
        s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        s.connect(("10.255.255.255", 1))
        ip = s.getsockname()[0]
        s.close()
        return ip
    except Exception:
        return "127.0.0.1"

import logging
import sys


def setup_logger(name: str = "liturgia_core") -> logging.Logger:
    logger = logging.getLogger(name)

    if not logger.handlers:
        logger.setLevel(logging.INFO)
        handler = logging.StreamHandler(sys.stdout)

        formatter = logging.Formatter(
            fmt='%(asctime)s | %(levelname)-8s | %(module)s:%(funcName)s:%(lineno)d | %(message)s',
            datefmt='%Y-%m-%d %H:%M:%S'
        )
        handler.setFormatter(formatter)
        logger.addHandler(handler)

    return logger


log = setup_logger()

    __tablename__ = "anuncios"
    id = Column(Integer, primary_key=True, index=True)
    titulo = Column(String, nullable=False)
    contenido = Column(Text, nullable=False)
    activo = Column(Boolean, default=True)

class TextoBiblico(Base):
    __tablename__ = "textos_biblicos"
    id = Column(Integer, primary_key=True, index=True)
    referencia = Column(String, index=True, nullable=False)

class Ofrenda(Base):

def normalize(text_val):
    if not text_val: return ""
    text_val = text_val.lower().strip()
    text_val = re.sub(r'^1\s*', '1', text_val)
    text_val = re.sub(r'^2\s*', '2', text_val)
    text_val = re.sub(r'^3\s*', '3', text_val)
    text_val = text_val.replace(" ", "")
    return ''.join(c for c in unicodedata.normalize('NFD', text_val) if unicodedata.category(c) != 'Mn')

def get_standard_book(search_term):
    norm_term = normalize(search_term)
    for standard, aliases in BIBLE_ALIASES.items():
        if norm_term == standard or standard.startswith(norm_term):
            return standard
        for a in aliases:
            if norm_term == a or a.startswith(norm_term):
                return standard
    return None

@router.get("/api/bible/search")
def search_bible(libro: str, capitulo: str, version: str = "RVR1960", db: Session = Depends(get_bible_db)):
    std_book = get_standard_book(libro)
    if not std_book:
        raise HTTPException(status_code=404, detail=f"[ERROR_LIBRO] Libro '{libro}' no reconocido.")

    chap_str = capitulo.strip()
    verse_str = None
    if ":" in capitulo:
        parts = capitulo.split(":", 1)
        chap_str = parts[0].strip()
        verse_str = parts[1].strip()

    try:
        chap_num = int(re.sub(r'\D', '', chap_str))
    except ValueError:
        raise HTTPException(status_code=400, detail="[ERROR_SINTAXIS] El capítulo debe ser un número válido.")

    try:
        check_query = text("SELECT 1 FROM versiculos WHERE version = :version AND libro = :libro AND capitulo = :capitulo LIMIT 1")
        exists = db.execute(check_query, {"version": version, "libro": std_book, "capitulo": chap_num}).fetchone()
        if not exists:
            raise HTTPException(status_code=404, detail=f"[ERROR_CAPÍTULO] El capítulo {chap_num} no existe en este libro.")

        if not verse_str:
            data_query = text("SELECT versiculo, texto, nombre_libro_original FROM versiculos WHERE version = :version AND libro = :libro AND capitulo = :capitulo ORDER BY versiculo")
            rows = db.execute(data_query, {"version": version, "libro": std_book, "capitulo": chap_num}).fetchall()

            final_text = "\n".join([f"{r[0]}. {r[1]}" for r in rows])
            ref_name = rows[0][2]
            referencia = f"{ref_name} {chap_num}"
        else:
            req_verses = []
            if "-" in verse_str:
                nums = re.findall(r'\d+', verse_str)
                if len(nums) >= 2:
                    req_verses = list(range(int(nums[0]), int(nums[1]) + 1))
                elif len(nums) == 1:
                    req_verses = [int(nums[0])]
            else:
                nums = re.findall(r'\d+', verse_str)
                if nums:
                    req_verses = [int(nums[0])]

            if not req_verses:
                raise HTTPException(status_code=400, detail="[ERROR_SINTAXIS] Formato de versículo inválido.")

            placeholders = ','.join(f':v{i}' for i in range(len(req_verses)))
            params = {"version": version, "libro": std_book, "capitulo": chap_num}
            for i, v in enumerate(req_verses):
                params[f"v{i}"] = v

            data_query = text(f"SELECT versiculo, texto, nombre_libro_original FROM versiculos WHERE version = :version AND libro = :libro AND capitulo = :capitulo AND versiculo IN ({placeholders}) ORDER BY versiculo")
            rows = db.execute(data_query, params).fetchall()

            if not rows:
                raise HTTPException(status_code=404, detail=f"[ERROR_VERSÍCULO] Versículo(s) {verse_str} no encontrado(s).")

            final_text = "\n".join([f"{r[0]}. {r[1]}" for r in rows])
            ref_name = rows[0][2]
            referencia = f"{ref_name} {chap_num}"

        return {
            "referencia": referencia,
            "version": version,
            "texto": final_text
        }
    except HTTPException as he:
        raise he
    except Exception as e:
        log.error(f"Error crítico en búsqueda bíblica: {str(e)}")
        raise HTTPException(status_code=500, detail="Error interno al procesar búsqueda bíblica")

class ConnectionManager:
    """Gestiona las conexiones WebSocket con aislamiento Preview / Program."""

    def __init__(self):
        self.live_projectors: list[WebSocket] = []
        self.preview_monitors: list[WebSocket] = []
        self.control_clients: list[WebSocket] = []

    async def connect_live(self, websocket: WebSocket):
        await websocket.accept()
        self.live_projectors.append(websocket)
        log.info(f"Proyector EN VIVO conectado. Total activos: {len(self.live_projectors)}")

    async def connect_preview(self, websocket: WebSocket):
        await websocket.accept()
        self.preview_monitors.append(websocket)
        log.info(f"Monitor PREVIEW conectado. Total activos: {len(self.preview_monitors)}")

    async def connect_control(self, websocket: WebSocket):
        await websocket.accept()
        self.control_clients.append(websocket)
        log.info(f"Cliente de control conectado. Total activos: {len(self.control_clients)}")

    def disconnect(self, websocket: WebSocket, client_type: str):
        if client_type == "live" and websocket in self.live_projectors:
            self.live_projectors.remove(websocket)
            log.info(f"Proyector EN VIVO desconectado. Restantes: {len(self.live_projectors)}")
        elif client_type == "preview" and websocket in self.preview_monitors:
            self.preview_monitors.remove(websocket)
            log.info(f"Monitor PREVIEW desconectado. Restantes: {len(self.preview_monitors)}")
        elif client_type == "control" and websocket in self.control_clients:
            self.control_clients.remove(websocket)
            log.info(f"Cliente de control desconectado. Restantes: {len(self.control_clients)}")

    async def send_to_live(self, message: dict):
        for client in self.live_projectors[:]:
            try:
                await client.send_json(message)
            except Exception as e:
                log.error(f"Fallo de transmisión a Proyector LIVE: {str(e)}")
                self.disconnect(client, "live")

    async def send_to_preview(self, message: dict):
        for client in self.preview_monitors[:]:
            try:
                await client.send_json(message)
            except Exception as e:
                log.error(f"Fallo de transmisión a Monitor PREVIEW: {str(e)}")
                self.disconnect(client, "preview")


manager = ConnectionManager()


@router.websocket("/ws/{client_type}")
async def websocket_endpoint(websocket: WebSocket, client_type: str):
    if client_type == "live":
        await manager.connect_live(websocket)
        try:
            while True:
                await websocket.receive_text()  # Keepalive
        except WebSocketDisconnect:
            manager.disconnect(websocket, "live")
        except Exception as e:
            log.error(f"Excepción en WS LIVE: {str(e)}")
            manager.disconnect(websocket, "live")

    elif client_type == "preview":
        await manager.connect_preview(websocket)
        try:
            while True:
                await websocket.receive_text()  # Keepalive
        except WebSocketDisconnect:
            manager.disconnect(websocket, "preview")
        except Exception as e:
            log.error(f"Excepción en WS PREVIEW: {str(e)}")
            manager.disconnect(websocket, "preview")

    elif client_type == "control":
        await manager.connect_control(websocket)
        try:
            while True:
                data = await websocket.receive_json()
                # El motor de enrutamiento: Decide a qué canal enviar la señal
                target = data.get("target", "all")

                if target == "live":
                    await manager.send_to_live(data)
                elif target == "preview":
                    await manager.send_to_preview(data)
                else:
                    # Comandos de emergencia o globales (clear, blackout) van a ambas pantallas
                    await manager.send_to_live(data)
                    await manager.send_to_preview(data)

        except WebSocketDisconnect:
            manager.disconnect(websocket, "control")
        except Exception as e:
            log.error(f"Excepción en WS CONTROL: {str(e)}")
            manager.disconnect(websocket, "control")

from sqlalchemy import extract
from sqlalchemy.orm import Session
from datetime import datetime, timedelta
from database.models import Cumpleanos
from core.logger import log
from schemas.admin_schemas import CumpleanosCreate


def obtener_cumpleanos_rango(db: Session):
    """
    Obtiene los cumpleaños en una ventana estricta de 7 días:
    Desde el miércoles pasado hasta el martes siguiente.
    """
    try:
        hoy = datetime.now()
        # En Python: Lunes=0, Martes=1, Miércoles=2, Jueves=3, Viernes=4, Sábado=5, Domingo=6
        # Fórmula para hallar cuántos días restar para llegar al miércoles más cercano hacia atrás
        dias_desde_miercoles = (hoy.weekday() - 2) % 7
        fecha_inicio = hoy - timedelta(days=dias_desde_miercoles)

        resultados = []

        # Iterar exactamente los 7 días (Miércoles [0] a Martes [6])
        for i in range(7):
            fecha_objetivo = fecha_inicio + timedelta(days=i)
            mes = fecha_objetivo.month
            dia = fecha_objetivo.day

            cumpleaneros = db.query(Cumpleanos).filter(
                extract('month', Cumpleanos.fecha_nacimiento) == mes,
                extract('day', Cumpleanos.fecha_nacimiento) == dia
            ).all()

            for c in cumpleaneros:
                # Calcula si es exactamente hoy o mañana para las etiquetas visuales
                diff_dias = (fecha_objetivo.date() - hoy.date()).days

                resultados.append({
                    "id": c.id,
                    "nombre": c.nombre,
                    "fecha": fecha_objetivo.strftime("%d/%m"),
                    "dias_diff": diff_dias,
                    "es_hoy": diff_dias == 0,
                    "es_manana": diff_dias == 1
                })

        # Retorna la lista en el orden natural cronológico (Miércoles a Martes)
        return resultados
    except Exception as e:
        log.error(f"Error al obtener cumpleaños en la ventana semanal: {str(e)}")
        return []

            query = query.filter(Ofrenda.fecha >= fecha_inicio)
        if fecha_fin:
            query = query.filter(Ofrenda.fecha <= fecha_fin)

        ofrendas = query.order_by(Ofrenda.fecha.desc()).all()
        return [
            {
                "id": o.id,
                "tipo": o.tipo,
                "monto": o.monto,
                "descripcion": o.descripcion,
                "fecha": o.fecha.strftime("%Y-%m-%d"),
                "anonima": o.anonima
            }
            for o in ofrendas
        ]
    except Exception as e:
        log.error(f"Error al obtener ofrendas: {str(e)}")
        return []


def obtener_total_ofrendas(db: Session, fecha_inicio: date = None, fecha_fin: date = None):
    """Obtiene el total monetario de ofrendas en un rango de fechas."""
    try:
        query = db.query(Ofrenda).filter(Ofrenda.monto != None)

        if fecha_inicio:
            query = query.filter(Ofrenda.fecha >= fecha_inicio)
        if fecha_fin:
            query = query.filter(Ofrenda.fecha <= fecha_fin)

        total = sum(o.monto for o in query.all() if o.monto)
        return {"total": total, "moneda": "USD"}
    except Exception as e:
        log.error(f"Error al calcular total de ofrendas: {str(e)}")
        return {"total": 0, "moneda": "USD"}


def crear_ofrenda(db: Session, data: OfrendaCreate):
    """Crea un nuevo registro de ofrenda."""
    try:
        fecha_obj = datetime.strptime(data.fecha, "%Y-%m-%d").date() if data.fecha else date.today()

        nueva = Ofrenda(
            tipo=data.tipo.strip(),
            monto=data.monto,
            descripcion=data.descripcion.strip() if data.descripcion else None,
            fecha=fecha_obj,
            anonima=data.anonima
        )
        db.add(nueva)
        db.commit()
        db.refresh(nueva)
        return {
            "id": nueva.id,
            "tipo": nueva.tipo,
            "monto": nueva.monto,
            "descripcion": nueva.descripcion,
            "fecha": nueva.fecha.strftime("%Y-%m-%d"),
            "anonima": nueva.anonima
        }
    except Exception as e:
        db.rollback()
        log.error(f"Error al crear ofrenda: {str(e)}")
        raise


def eliminar_ofrenda(db: Session, id: int):
    """Elimina una ofrenda por ID."""
    try:
        ofrenda = db.query(Ofrenda).filter(Ofrenda.id == id).first()
        if ofrenda:
            db.delete(ofrenda)
            db.commit()
            return True
        return False
    except Exception as e:
        db.rollback()
        log.error(f"Error al eliminar ofrenda ID {id}: {str(e)}")
        return False
